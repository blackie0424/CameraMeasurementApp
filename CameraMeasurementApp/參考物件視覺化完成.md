## 程式碼實作說明

### 主要方法

#### 1. createAnnotatedImage

```swift
private func createAnnotatedImage(from record: MeasurementRecord) -> UIImage {
    let renderer = UIGraphicsImageRenderer(size: record.image.size)

    return renderer.image { context in
        // 1. 繪製原始影像
        record.image.draw(at: .zero)

        let ctx = context.cgContext

        // 2. 繪製測量物體標註（綠色）
        for object in record.detectedObjects {
            // 邊界框和數值
        }

        // 3. 繪製參考物件視覺化（橘色）
        if let reference = record.referenceObject ?? selectedReferenceObject {
            drawReferenceObject(reference, in: ctx, imageSize: record.image.size)
        }
    }
}
```

#### 2. drawReferenceObject

```swift
private func drawReferenceObject(_ reference: ReferenceObject,
                                in context: CGContext,
                                imageSize: CGSize) {
    // 1. 計算位置（右下角）
    let refRect = CGRect(
        x: imageSize.width - 120 - 20,
        y: imageSize.height - 120 - 20,
        width: 120,
        height: 120
    )

    // 2. 繪製背景和邊框
    context.setFillColor(UIColor.black.withAlphaComponent(0.6).cgColor)
    context.fill(refRect)
    context.setStrokeColor(UIColor.systemOrange.cgColor)
    context.stroke(refRect)

    // 3. 繪製圖示
    drawReferenceIcon(for: reference, in: iconRect, context: context)

    // 4. 繪製文字標註
    // 名稱和尺寸
}
```

#### 3. drawReferenceIcon

```swift
private func drawReferenceIcon(for reference: ReferenceObject,
                              in rect: CGRect,
                              context: CGContext) {
    // 根據參考物件類型繪製不同形狀
    switch reference.name {
    case "打火機": // 矩形
    case "硬幣":   // 圓形
    case "信用卡": // 圓角矩形
    case "iPhone": // 手機形狀
    case "原子筆": // 細長矩形
    }
}
```

## 技術細節

### 圖形繪製

使用 Core Graphics 進行繪製：

```swift
// 矩形
context.fill(rect)
context.stroke(rect)

// 圓形
context.fillEllipse(in: rect)
context.strokeEllipse(in: rect)

// 圓角矩形
let path = UIBezierPath(roundedRect: rect, cornerRadius: radius)
context.addPath(path.cgPath)
context.fillPath()
```

### 文字繪製

使用 NSAttributedString 進行文字繪製：

```swift
let attributes: [NSAttributedString.Key: Any] = [
    .font: UIFont.systemFont(ofSize: 14, weight: .bold),
    .foregroundColor: UIColor.white
]
text.draw(at: point, withAttributes: attributes)
```

### 半透明效果

使用 alpha 通道創建半透明背景：

```swift
UIColor.black.withAlphaComponent(0.6)  // 60% 透明度
```
